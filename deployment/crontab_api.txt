# Set timezone to Norway (Europe/Oslo)
TZ=Europe/Oslo

# SNKRDUNK scan every 6 hours at 0, 6, 12, 18
# Logs viewable at: http://localhost:8000 → SNKRDUNK section
0 6,12,18,0 * * * curl -s -X POST "http://localhost:8000/api/v1/snkrdunk/fetch" -H "Content-Type: application/json" -d '{"pages":[1],"force_refresh":true}' >> ~/logs/snkrdunk_api.log 2>&1

# Competitor price refresh every 6 hours at :30 (pulls latest prices from MarketIntel)
30 6,12,18,0 * * * curl -s -X POST "http://localhost:8000/api/v1/competitor-links/refresh-prices" >> ~/logs/competitor_refresh.log 2>&1

# Shopify collection sync + clear expired stock dates - daily at 1:00 AM
# Syncs Pokemon collection and clears any stock dates that have arrived
0 1 * * * cd ~/shopify && python3 scripts/cron_shopify_sync.py >> ~/logs/shopify_sync_api.log 2>&1

# Daily cleanup: Remove old log files (older than 30 days)
0 3 * * * find ~/logs -name "*.log" -type f -mtime +30 -delete
